212 def build_placeholder_question
213   base_slots = @template.persisted? ? 1 : 3
214   active_questions = @template.template_questions.reject(&:marked_for_destruction?).size
215   total_needed = [active_questions, base_slots].max
216 
217   while active_questions < total_needed
218     @template.template_questions.build(
219       question_type: TemplateQuestion::QUESTION_TYPES[:likert],
220       position: @template.template_questions.size + 1,
221       required: true,
222       min_value: 1,
223       max_value: 5
224     )
225     active_questions += 1
226   end
227 end
