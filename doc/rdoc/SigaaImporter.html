    class SigaaImporter

    app/services/sigaa_importer.rb

    Service responsible for importing data from SIGAA JSON files into the system. Handles creation and update of subjects (matérias), classes (turmas), teachers (docentes), students (dicentes), and enrollments (matrículas).

    Example:

    result = SigaaImporter.call(
  classes_file: File.open('turmas.json'),
  class_members_file: File.open('turma_docente_dicentes.json')
)
puts result.summary_message

277 def import_docente_dicentes_and_matriculas(member_data)
278   materia = Materia.find_by(code: member_data['code'])
279   unless materia
280     @result.errors << "Matéria #{member_data['code']} não encontrada"
281     return
282   end
283 
284   turma = Turma.find_by(
285     materia: materia,
286     class_code: member_data['classCode'],
287     semester: member_data['semester']
288   )
289   
290   unless turma
291     @result.errors << "Turma #{member_data['code']}-#{member_data['classCode']} não encontrada"
292     return
293   end
294 
295   # Importar docente
296   if member_data['docente']
297     docente = import_docente(member_data['docente'])
298     if docente && turma.docente != docente
299       turma.update(docente: docente)
300       @result.updated[:turmas] += 1
301     end
302   end
303 
304   # Importar dicentes
305   if member_data['dicente']
306     member_data['dicente'].each do |dicente_data|
307       import_dicente_and_matricula(dicente_data, turma)
308     end
309   end
310 end
